% Part heading  (I. / II. / III.)
\newcommand{\partheading}[1]{%
  \noindent\textbf{#1}\par\vspace{1pt}%
}

% Subsection heading (a. / b. / c. ...)
% 3pt before — enough visual separation without wasting a full line.
\newcommand{\nsercsubheading}[1]{%
  \Needspace{2\baselineskip}%
  \vspace{3pt}%
  \noindent\textbf{#1}\par\vspace{0pt}%
}

% Numbered citation with hanging indent + italic role note on a second indented line.
% Inter-item gap 2pt: perceptibly more than the 0pt between wrapped citation lines.
\newcommand{\nsercitem}[2]{%
  \noindent\hangindent=1.5em\hangafter=1 #1\par%
  \noindent\hspace{1.5em}\textit{#2}\par%
  \vspace{2pt}%
}

% Inline None entry — used for sections e and f to avoid wasting a blank line.
\newcommand{\nsercnone}[1]{%
  \Needspace{1\baselineskip}%
  \vspace{3pt}%
  \noindent\textbf{#1} None.\par%
}
